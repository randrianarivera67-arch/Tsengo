function sceneMarkup(){
  const G={cell:cell,pad:pad,W:W,H:H,pts:PTS};
  const ah=activeHuman(),R=cell*0.315,out=[];
  const hl={},ct={},capturable={};
  for(var a=0;a<s.availableOptions.length;a++){hl[s.availableOptions[a].to]=1;for(var b=0;b<s.availableOptions[a].captured.length;b++)ct[s.availableOptions[a].captured[b]]=1;}
  if(!s.winner&&s.toMove===ah){const all=getAllTurns(s.board,ah);for(var i1=0;i1<all.length;i1++)if(all[i1].path[0].captured.length)capturable[all[i1].path[0].from]=1;}
  const acc=ACC[ah];
  out.push('<g pointer-events="none">');
  out.push('<rect x="6" y="6" width="'+(G.W-12)+'" height="'+(G.H-12)+'" rx="22" fill="url(#boardG)" filter="url(#slabSh)"/>');
  out.push('<rect x="9" y="9" width="'+(G.W-18)+'" height="'+(G.H-18)+'" rx="19" fill="none" stroke="rgba(255,255,255,.08)" stroke-width="2"/>');
  for(var i=0;i<SIZE;i++){
    const p1=G.pts[i],q=rc(i);
    for(var k=0;k<NDIRS[i].length;k++){
      const ni=idx(q[0]+NDIRS[i][k][0],q[1]+NDIRS[i][k][1]);
      if(ni<i)continue;
      const p2=G.pts[ni];
      out.push('<line class="gline" x1="'+p1.x+'" y1="'+p1.y+'" x2="'+p2.x+'" y2="'+p2.y+'" stroke-width="1.5" opacity=".45" stroke-linecap="round"/>');
    }
  }
  for(var k3=0;k3+1<s.trail.length;k3++){
    const t1=G.pts[s.trail[k3]],t2=G.pts[s.trail[k3+1]];
    out.push('<line x1="'+t1.x+'" y1="'+t1.y+'" x2="'+t2.x+'" y2="'+t2.y+'" stroke="#F0C36D" stroke-width="4" stroke-linecap="round" opacity=".85"/>');
  }
  for(var i2=0;i2<G.pts.length;i2++){out.push('<circle class="gdot" cx="'+G.pts[i2].x+'" cy="'+G.pts[i2].y+'" r="5" opacity=".55"/>');}
  for(var i3=0;i3<G.pts.length;i3++){out.push('<text class="glbl" x="'+G.pts[i3].x+'" y="'+(G.pts[i3].y-cell*0.44)+'" text-anchor="middle" font-size="9" font-family="\'IBM Plex Mono\',monospace" opacity=".55">'+pointLabel(G.pts[i3].i)+"</text>");}
  for(var i4=0;i4<G.pts.length;i4++){if(hl[G.pts[i4].i]){out.push('<circle cx="'+G.pts[i4].x+'" cy="'+G.pts[i4].y+'" r="8" fill="'+acc.bead+'" stroke="'+acc.rim+'" stroke-width="1" class="pulse"/>');}}
  out.push('</g><g pointer-events="none">');
  for(var i5=0;i5<G.pts.length;i5++){if(ct[G.pts[i5].i]){out.push('<circle cx="'+G.pts[i5].x+'" cy="'+G.pts[i5].y+'" r="'+(R*1.18)+'" fill="none" stroke="#E5484D" stroke-width="3" stroke-dasharray="7 6" class="pulse"/>');}}
  out.push("</g>");
  for(var i6=0;i6<G.pts.length;i6++){
    const p6=G.pts[i6],v=s.board[p6.i];if(!v)continue;
    const sel=s.selected===p6.i,must=capturable[p6.i]&&!sel;
    const mine=v===ah&&s.toMove===ah&&!s.thinking&&!s.winner;
    const sa=ACC[v];
    out.push('<g pointer-events="none">');
    out.push('<ellipse cx="'+p6.x+'" cy="'+(p6.y+R*0.82)+'" rx="'+(R*0.92)+'" ry="'+(R*0.36)+'" fill="rgba(0,0,0,.5)" filter="url(#blur3)"/>');
    if(must)out.push('<circle cx="'+p6.x+'" cy="'+p6.y+'" r="'+(R*1.32)+'" fill="none" stroke="#E5484D" stroke-width="2.5" class="pulse"/>');
    if(sel){out.push('<circle cx="'+p6.x+'" cy="'+p6.y+'" r="'+(R*1.38)+'" fill="none" stroke="'+sa.ring+'" stroke-width="3" class="pulse"/>');out.push('<circle cx="'+p6.x+'" cy="'+p6.y+'" r="'+(R*1.55)+'" fill="none" stroke="'+sa.ring+'" stroke-width="1.5" opacity=".35"/>');}
    out.push("</g>");
    out.push('<g class="stone'+(mine?" mine":"")+'" data-i="'+p6.i+'">');
    out.push('<circle cx="'+p6.x+'" cy="'+p6.y+'" r="'+R+'" fill="'+(v==="W"?"url(#stW)":"url(#stB)")+'"/>');
    out.push('<circle cx="'+p6.x+'" cy="'+p6.y+'" r="'+R+'" fill="none" stroke="'+(v==="W"?"rgba(143,160,198,.7)":"rgba(0,0,0,.8)")+'" stroke-width="1"/>');
    out.push('<ellipse cx="'+(p6.x-R*0.32)+'" cy="'+(p6.y-R*0.4)+'" rx="'+(R*0.44)+'" ry="'+(R*0.3)+'" fill="url(#spec)" opacity="'+(v==="W"?0.95:0.5)+'"/>');
    out.push('<ellipse cx="'+(p6.x+R*0.2)+'" cy="'+(p6.y+R*0.55)+'" rx="'+(R*0.5)+'" ry="'+(R*0.22)+'" fill="url(#spec)" opacity="'+(v==="W"?0.35:0.15)+'"/>');
    out.push("</g>");
  }
  for(var i7=0;i7<G.pts.length;i7++){
    const p7=G.pts[i7];if(s.board[p7.i]!==null)continue;
    out.push('<circle data-i="'+p7.i+'" cx="'+p7.x+'" cy="'+p7.y+'" r="'+(cell*0.4)+'" fill="transparent" style="cursor:'+(hl[p7.i]?"pointer":"default")+'"/>');
  }
  return out.join("");
}
